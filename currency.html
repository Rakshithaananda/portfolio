Currency Converter (Python)

This project converts currencies using Python.


print("===== Currency Converter =====")

usd_to_inr = 83.0
inr_to_usd = 1 / usd_to_inr

amount = float(input("Enter amount: "))
from_currency = input("From currency (USD/INR): ").upper()
to_currency = input("To currency (USD/INR): ").upper()

if from_currency == "USD" and to_currency == "INR":
    result = amount * usd_to_inr
elif from_currency == "INR" and to_currency == "USD":
    result = amount * inr_to_usd
else:
    print("Conversion not available.")
    result = None

if result:
    print("Converted Amount:", round(result, 2), to_currency)
